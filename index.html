//@version=5 strategy("VN30F1M - MA20/50 Cross | SL nến/điểm | R:R | Trailing", overlay=true, initial_capital=100000000, default_qty_type=strategy.fixed, default_qty_value=1, commission_type=strategy.commission.percent, commission_value=0, pyramiding=0, calc_on_every_tick=false, process_orders_on_close=false) // ===== INPUTS ===== maFastLen = input.int(20, "MA nhanh", minval=1, group="Tín hiệu") maSlowLen = input.int(50, "MA chậm", minval=1, group="Tín hiệu") maType = input.string("SMA", "Loại MA", options=["SMA", "EMA", "WMA"], group="Tín hiệu") slMode = input.string("Điểm cố định", "Kiểu SL", options=["Đỉnh/đáy N nến", "Điểm cố định"], group="Quản lý lệnh") slLookback = input.int(9, "Số nến (nếu SL theo nến)", minval=1, group="Quản lý lệnh") slPoints = input.float(2.0, "SL (điểm, nếu SL cố định)", minval=0.1, step=0.5, group="Quản lý lệnh") rr = input.float(4.0, "R:R", minval=0.1, step=0.1, group="Quản lý lệnh") minSLpts = input.float(0.5, "SL tối thiểu (điểm, chỉ SL theo nến)", minval=0.0, step=0.1, group="Quản lý lệnh") maxSLpts = input.float(0.0, "SL tối đa (điểm, 0 = tắt, chỉ SL theo nến)", minval=0.0, step=0.1, group="Quản lý lệnh") // ===== ĐƠN VỊ: ĐIỂM hay PHẦN TRĂM ===== unitMode = input.string("Tự động", "Đơn vị SL/trailing", options=["Tự động", "Điểm", "Phần trăm"], group="Đơn vị", tooltip="Tự động: cổ phiếu dùng %, các loại khác (phái sinh, chỉ số) dùng điểm") noShortStock = input.bool(true, "Tự tắt Short khi là cổ phiếu", group="Đơn vị", tooltip="Cổ phiếu Việt Nam không bán khống được") slPct = input.float(0.5, "SL (%, nếu dùng phần trăm)", minval=0.01, step=0.05, group="Đơn vị") minSLpct = input.float(0.1, "SL tối thiểu (%, chỉ SL theo nến)", minval=0.0, step=0.05, group="Đơn vị") maxSLpct = input.float(0.0, "SL tối đa (%, 0 = tắt, chỉ SL theo nến)", minval=0.0, step=0.05, group="Đơn vị") trailStartPct = input.float(0.5, "Trailing: kích hoạt khi lời (%)", minval=0.01, step=0.05, group="Đơn vị") lockPct = input.float(0.0, "Trailing: khóa lời lần đầu (%)", step=0.05, group="Đơn vị") trailStepPct = input.float(0.5, "Trailing: bước nâng (%)", minval=0.01, step=0.05, group="Đơn vị") stockCapital = input.float(100000000, "Vốn mỗi lệnh khi dùng % / cổ phiếu (VND)", minval=1000000, step=1000000, group="Đơn vị", tooltip="Khối lượng = vốn / giá, làm tròn xuống theo lô. Phái sinh luôn vào 1 hợp đồng") lotSize = input.int(100, "Lô giao dịch (cổ phiếu)", minval=1, group="Đơn vị") isStock = syminfo.type == "stock" usePct = unitMode == "Phần trăm" or (unitMode == "Tự động" and isStock) useTrail = input.bool(true, "Bật trailing (nâng SL)", group="Trailing") trailStart = input.float(2.0, "Kích hoạt khi lời (điểm)", minval=0.1, step=0.5, group="Trailing", tooltip="Khi lợi nhuận tối đa đạt mức này, SL được nâng lần đầu") lockPts = input.float(0.0, "Khóa lời lần đầu (điểm, 0 = hòa vốn)", step=0.1, group="Trailing", tooltip="SL nâng lên = giá vào + số điểm này. Nên đặt nhỉnh hơn 0 để bù phí") trailStep = input.float(2.0, "Bước nâng (điểm)", minval=0.1, step=0.5, group="Trailing", tooltip="Cứ lời thêm mỗi bước này thì SL nâng thêm đúng bước đó. Đặt nhỏ = bám sát giá") removeTP = input.bool(false, "Bỏ TP khi trailing (gồng không giới hạn)", group="Trailing", tooltip="Bật: không chốt lời cố định, chỉ thoát khi SL (đã nâng) bị chạm hoặc cuối phiên") allowLong = input.bool(true, "Cho phép Long", group="Hướng lệnh") allowShort = input.bool(true, "Cho phép Short", group="Hướng lệnh") useSession = input.bool(true, "Chỉ vào lệnh trong phiên", group="Thời gian") sessionStr = input.session("0900-1425", "Phiên vào lệnh (GMT+7)", group="Thời gian") closeEOD = input.bool(true, "Đóng lệnh cuối phiên", group="Thời gian") closeTime = input.session("1440-1445", "Khung giờ đóng lệnh", group="Thời gian") // ===== MA ===== ma(src, len) => switch maType "SMA" => ta.sma(src, len) "EMA" => ta.ema(src, len) "WMA" => ta.wma(src, len) maFast = ma(close, maFastLen) maSlow = ma(close, maSlowLen) plot(maFast, "MA nhanh", color=color.orange, linewidth=2) plot(maSlow, "MA chậm", color=color.blue, linewidth=2) // ===== TÍN HIỆU ===== crossUp = ta.crossover(maFast, maSlow) crossDown = ta.crossunder(maFast, maSlow) inSession = not useSession or not na(time(timeframe.period, sessionStr, "Asia/Ho_Chi_Minh")) isCloseTime = closeEOD and not na(time(timeframe.period, closeTime, "Asia/Ho_Chi_Minh")) useCandleSL = slMode == "Đỉnh/đáy N nến" lowestPrev = ta.lowest(low, slLookback)[1] highestPrev = ta.highest(high, slLookback)[1] longRiskEst = close - lowestPrev shortRiskEst = highestPrev - close minSLeff = usePct ? minSLpct / 100.0 * close : minSLpts maxSLeff = usePct ? maxSLpct / 100.0 * close : maxSLpts longRiskOK = not useCandleSL or (longRiskEst >= minSLeff and (maxSLeff == 0 or longRiskEst <= maxSLeff)) shortRiskOK = not useCandleSL or (shortRiskEst >= minSLeff and (maxSLeff == 0 or shortRiskEst <= maxSLeff)) longOK = allowLong and crossUp and inSession and longRiskOK shortOK = allowShort and crossDown and inSession and shortRiskOK and not (noShortStock and isStock) // ===== VÀO LỆNH ===== // Chế độ % (cổ phiếu): khối lượng theo vốn cố định. Phái sinh: qty = na -> dùng 1 hợp đồng mặc định stockQty = math.floor(stockCapital / close / lotSize) * lotSize qtyArg = usePct ? float(stockQty) : na var float pendingSL = na var float slPrice = na var float tpPrice = na var float entryPx = na var float bestPx = na // giá thuận lợi nhất kể từ khi vào lệnh var float trStartEff = na // các ngưỡng trailing đã quy ra điểm giá tại lúc vào lệnh var float lockEff = na var float stepEff = na if longOK and strategy.position_size == 0 and (not usePct or stockQty > 0) strategy.entry("Long", strategy.long, qty=qtyArg) pendingSL := lowestPrev if shortOK and strategy.position_size == 0 and (not usePct or stockQty > 0) strategy.entry("Short", strategy.short, qty=qtyArg) pendingSL := highestPrev // ===== TÍNH SL/TP THEO GIÁ KHỚP THỰC TẾ ===== newTrade = strategy.position_size != 0 and strategy.position_size[1] == 0 if newTrade entryPx := strategy.position_avg_price isLong = strategy.position_size > 0 k = usePct ? entryPx / 100.0 : 1.0 // 1% giá vào = k điểm giá (chế độ điểm: k = 1) float risk = na if useCandleSL risk := math.abs(entryPx - pendingSL) else risk := (usePct ? slPct : slPoints) * k trStartEff := (usePct ? trailStartPct : trailStart) * k lockEff := (usePct ? lockPct : lockPts) * k stepEff := math.max((usePct ? trailStepPct : trailStep) * k, syminfo.mintick) risk := math.max(risk, syminfo.mintick) slPrice := isLong ? entryPx - risk : entryPx + risk tpPrice := isLong ? entryPx + risk * rr : entryPx - risk * rr bestPx := entryPx // ===== TRAILING: NÂNG SL THEO BẬC ===== if useTrail and strategy.position_size != 0 if strategy.position_size > 0 bestPx := math.max(bestPx, high) profit = bestPx - entryPx if profit >= trStartEff steps = math.floor((profit - trStartEff) / stepEff) newSL = entryPx + lockEff + steps * stepEff slPrice := math.max(slPrice, newSL) // Long: chỉ nâng SL lên else bestPx := math.min(bestPx, low) profit = entryPx - bestPx if profit >= trStartEff steps = math.floor((profit - trStartEff) / stepEff) newSL = entryPx - lockEff - steps * stepEff slPrice := math.min(slPrice, newSL) // Short: chỉ hạ SL xuống // ===== THOÁT LỆNH ===== tpUse = (useTrail and removeTP) ? na : tpPrice if strategy.position_size > 0 strategy.exit("Exit L", "Long", stop=slPrice, limit=tpUse) if strategy.position_size < 0 strategy.exit("Exit S", "Short", stop=slPrice, limit=tpUse) if isCloseTime strategy.close_all(comment="EOD") if strategy.position_size == 0 slPrice := na tpPrice := na bestPx := na // ===== PHÍ + THUẾ: TỰ NHẬN THEO LOẠI SẢN PHẨM (mã tự tính, xem trong Data Window) ===== stockFeeSide = input.float(0.3, "CP: phí mỗi chiều (%)", minval=0.0, step=0.05, group="Phí & thuế") stockTax = input.float(0.1, "CP: thuế bán (%)", minval=0.0, step=0.05, group="Phí & thuế") futRoundTrip = input.float(0.5, "Phái sinh: phí + thuế cả vòng (điểm)", minval=0.0, step=0.1, group="Phí & thuế") futPointVnd = input.float(100000, "Phái sinh: VND mỗi điểm", minval=1, group="Phí & thuế") var float netPnl = 0.0 var float feeSum = 0.0 var int winCnt = 0 var int doneCnt = 0 while doneCnt < strategy.closedtrades ep = strategy.closedtrades.entry_price(doneCnt) xp = strategy.closedtrades.exit_price(doneCnt) qs = strategy.closedtrades.size(doneCnt) aq = math.abs(qs) dirn = qs > 0 ? 1.0 : -1.0 float gross = na float fee = na if usePct gross := (xp - ep) * dirn * aq fee := (ep + xp) * aq * stockFeeSide / 100.0 + xp * aq * stockTax / 100.0 else gross := (xp - ep) * dirn * aq * futPointVnd fee := futRoundTrip * aq * futPointVnd netTrade = gross - fee netPnl += netTrade feeSum += fee winCnt += netTrade > 0 ? 1 : 0 doneCnt += 1 plot(netPnl, "Lãi ròng sau phí (VND)", display=display.data_window) plot(feeSum, "Tổng phí + thuế (VND)", display=display.data_window) // ===== HIỂN THỊ ===== plot(slPrice, "SL", color=color.red, style=plot.style_linebr) plot(tpUse, "TP", color=color.green, style=plot.style_linebr) plotshape(longOK and strategy.position_size == 0, "Long", shape.triangleup, location.belowbar, color.green, size=size.tiny) plotshape(shortOK and strategy.position_size == 0, "Short", shape.triangledown, location.abovebar, color.red, size=size.tiny)
